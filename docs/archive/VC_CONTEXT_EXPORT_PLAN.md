# VC Context Export to PDF/TXT - Implementation Plan

## 🎯 Goal
Enable VCs to **export the AI-generated summary** from VC Context Manager as a **PDF or TXT document** that can be sent along with pitch deck analysis.

## 📋 Current State
✅ **Already Implemented:**
- VC Context Manager uploads documents (notes, emails, meeting transcripts)
- AI synthesizes context using Gemini
- Context exports to Deck Intelligence for enhanced analysis
- Context stored in localStorage

## 🆕 New Feature: Export Summary as PDF/TXT

### Use Cases
1. **Share with Team** - Send summary PDF to other VCs/partners
2. **Attach to Deck** - Bundle summary with pitch deck for comprehensive review
3. **Due Diligence** - Include in DD documentation package
4. **Follow-up** - Send to founders as meeting recap/action items

## 🏗️ Architecture

```
VC Context Manager
    ↓
AI generates summary
    ↓
[Export as PDF] [Export as TXT] buttons
    ↓
Backend generates formatted document
    ↓
Download starts automatically
```

## 📦 Implementation Details

### Frontend API Call
```typescript
// src/services/vcContextApi.ts
async downloadReport(deckId: string, format: 'pdf' | 'txt'): Promise<Blob> {
  const response = await fetch(
    `${API_URL}/vc-context/${deckId}/export/${format}`
  );
  
  if (!response.ok) throw new Error('Download failed');
  
  return await response.blob();
}

// Trigger download in browser
const blob = await vcContextApi.downloadReport(deckId, 'pdf');
const url = window.URL.createObjectURL(blob);
const a = document.createElement('a');
a.href = url;
a.download = `vc-context-${companyName}-${Date.now()}.pdf`;
a.click();
window.URL.revokeObjectURL(url);
```

### Backend Route Handler
```typescript
// GET /api/vc-context/:deckId/export/pdf
router.get('/:deckId/export/pdf', async (req, res) => {
  const { deckId } = req.params;
  
  // Fetch summary and items
  const summary = await getLatestSummary(deckId);
  const items = await getContextItems(deckId);
  
  // Generate PDF
  const pdfBuffer = await generateVCContextPDF(
    deckId,
    summary.companyName,
    summary,
    items
  );
  
  // Send as download
  res.setHeader('Content-Type', 'application/pdf');
  res.setHeader(
    'Content-Disposition',
    `attachment; filename="vc-context-summary-${deckId}.pdf"`
  );
  res.send(pdfBuffer);
});
```

### PDF Generation with PDFKit
```typescript
import PDFDocument from 'pdfkit';

export async function generateVCContextPDF(
  deckId: string,
  companyName: string,
  summary: ContextSummary,
  items: ContextItem[]
): Promise<Buffer> {
  return new Promise((resolve, reject) => {
    const doc = new PDFDocument({ margin: 50 });
    const chunks: Buffer[] = [];
    
    doc.on('data', (chunk) => chunks.push(chunk));
    doc.on('end', () => resolve(Buffer.concat(chunks)));
    doc.on('error', reject);
    
    // Header
    doc.fontSize(24).text('VC Context Summary', { align: 'center' });
    doc.moveDown();
    doc.fontSize(14).text(`Company: ${companyName}`, { align: 'center' });
    doc.fontSize(10).text(
      `Generated: ${new Date().toLocaleString()}`,
      { align: 'center' }
    );
    doc.moveDown(2);
    
    // Executive Summary
    doc.fontSize(16).text('EXECUTIVE SUMMARY', { underline: true });
    doc.moveDown();
    doc.fontSize(11).text(summary.executiveSummary);
    doc.moveDown(2);
    
    // Key Insights
    doc.fontSize(16).text('KEY INSIGHTS', { underline: true });
    doc.moveDown();
    summary.keyInsights.forEach((insight, idx) => {
      doc.fontSize(11).text(`• ${insight}`, { indent: 20 });
      doc.moveDown(0.5);
    });
    doc.moveDown();
    
    // Opportunities
    doc.fontSize(16).text('OPPORTUNITIES', { underline: true });
    doc.moveDown();
    summary.opportunities.forEach((opp, idx) => {
      doc.fontSize(11).text(`• ${opp}`, { indent: 20 });
      doc.moveDown(0.5);
    });
    doc.moveDown();
    
    // Risks
    doc.fontSize(16).text('RISKS & CONCERNS', { underline: true });
    doc.moveDown();
    summary.risks.forEach((risk, idx) => {
      doc.fontSize(11).text(`• ${risk}`, { indent: 20 });
      doc.moveDown(0.5);
    });
    doc.moveDown();
    
    // Team Assessment
    doc.fontSize(16).text('TEAM ASSESSMENT', { underline: true });
    doc.moveDown();
    doc.fontSize(11).text(summary.teamAssessment);
    doc.moveDown(2);
    
    // Next Steps
    doc.fontSize(16).text('NEXT STEPS', { underline: true });
    doc.moveDown();
    summary.nextSteps.forEach((step, idx) => {
      doc.fontSize(11).text(`${idx + 1}. ${step}`, { indent: 20 });
      doc.moveDown(0.5);
    });
    doc.moveDown();
    
    // Recommendation
    doc.fontSize(16).text('INVESTMENT RECOMMENDATION', { underline: true });
    doc.moveDown();
    doc.fontSize(12).fillColor('green').text(
      `Decision: ${summary.recommendation.decision.toUpperCase()}`,
      { bold: true }
    );
    doc.fillColor('black');
    doc.fontSize(10).text(
      `Confidence: ${summary.recommendation.confidence}%`
    );
    doc.moveDown();
    doc.fontSize(11).text(summary.recommendation.rationale);
    doc.moveDown(2);
    
    // Source Documents
    doc.fontSize(16).text('SOURCE DOCUMENTS', { underline: true });
    doc.moveDown();
    items.forEach((item, idx) => {
      const date = new Date(item.created_at).toLocaleDateString();
      doc.fontSize(10).text(
        `${idx + 1}. ${item.file_name} (${date})`,
        { indent: 20 }
      );
      doc.moveDown(0.3);
    });
    
    // Footer
    doc.moveDown(2);
    doc.fontSize(8).fillColor('gray').text(
      'Generated by Startup Scout AI',
      { align: 'center' }
    );
    
    doc.end();
  });
}
```
